>> 03-comparacion.js
-------------------------------------------
let a = 10;

// Operadores deRelacionales
console.log(a > 5); // si es mayor
console.log(a >= 5); // si es mayor o igual
console.log(a < 5); // si es menor
console.log(a <= 5); // si es menor o igual

// Operadores de Igualdad
console.log(a == 10); // si a es igual a 10
console.log(a != 10); // si a es diferente a 10
console.log(a == "10"); // con == pregunta si el valor es igual y no el tipo de dato
console.log(a === 10); // preguntar si es igual el valor y tipo de dato
console.log(a !== 10); // preguntar si es diferente el valor y el tipo de dato

// Los operadores lógicos son AND, OR, NOT

// AND (&&)
let mayor = false;
let suscrito = true;
// console.log(true && true); // el AND revisa si TODOS los condicionales son verdaderos
// console.log(false && true); // Si un condicional es false con AND, se devuelve false
console.log("AND ", mayor && suscrito);

// OR (||)
console.log("OR ", mayor || suscrito); // el OR revisa si un solo condicional es verdadero. Aunque el resto sea false y uno sea true. Este devolvera true

// NOT (!)
console.log("NOT ", !mayor); // el NOT revisa si un condicional es falso. Si en este caso, el condicional es falso. Este devuelve true

// Byte 0000 0101 -> 5
// Byte 0000 0110 -> 6
// Byte 0000 0111 -> 7

console.log(3 | 5);
