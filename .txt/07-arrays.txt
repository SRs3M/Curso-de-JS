const usuarios = [
    {id:1, name: "Chanchito"},
    {id:1, name:"Felipe"},
    {id:2, name: "Feliz"},
]

// Una funcion que tiene que le tienen que pasar una funcion como argumento, se conocen como predicate (funcion que devuelve true o false)
const ans = usuarios.find(function(usuario) { // find recibirá la funcion la cual recibe un parametro el cual es un elemento del array que estamos iterando
    return usuario.id === 1; // Como hay dos id iguales, Si quiero que me retorne el de Felipe, solo me retornará el de chanchtio ya que es el primero que encontró, por eso se debe asegurar que el objeto que vamos a buscar sea único

// Una forma mas simple de escribirlo con arrow functions
const respuesta = usuarios.find(usuario =>
    usuario.name === "Chanchito"
);

// Si por alguna razon necesitamos el índice de lo que buscamos y no el objeto en si usamos findIndex
const index = usuarios.findIndex(usuario =>
    usuario.name === "Chanchito"
);
console.log(index);

>> 07-combinando.js
-------------------------------------------
let arr1 = [1, 2];
let arr2 = [3, 4];

// Combinar arrays //
let combinados = arr1.concat(arr2); // Concat no cambia el valor del array, asi que toca poner este valor en otra varialbe
console.log(combinados, arr1, arr2);
// Dividir los arrays //
let divididos = combinados.slice(1, 3); // indice desde que se empezara a eliminar objetos, indice donde se terminará este proceso
console.log(combinados, divididos);

let tragados = combinados.slice(1); // Podemos saltar el primer valor para que tome el resto de valores
console.log(combinados, tragados);

let masomenosDivididos = combinados.slice(); // Hará una copia de el array, (si modificamos el array original, también modificaremos este)
console.log(combinados, masomenosDivididos);

// problema
let conMayusculas = ["Z", "a", "d"];
// conMayusculas.sort(); 
// esto nos retornará Z, a, d
// Esto pasa porque los caracteres se le asgina un numero el cual se usa en los lenguajes de operación o los mismos SO para ser referenciados, por la ASCII(American Standard Code for Information Interchange) la letra Z se le asigna el 90, pero la a tiene el 97. Asi que Z < a, por eso sort nos devuelve la Z primero, Y si queremos desde orden alfabetico y no por números, tendremos que usar otro metodo de sort
conMayusculas.sort((a, b) => { // Normalmente se pasa por parametro a y b
    // a antes de b => -1
    // b antes de a => 1
    // si son iguales => 0
    let alower = a.toLowerCase(); // no importa si lo ponemos en UpperCase, lo que importa es que todos esten o en mayuscula o en minuscula
    let blower = b.toLowerCase();

    if(alower < blower) {
        return -1; // b antes de a (a, b)
    } // no es necesario poner un else, porque si el if de arriba es verdadero, hace un return y se corta la función que estamos haciendo
    if(alower > blower) {
        return 1;
    }
    return 0; // Seria como un else {return 0;}
});
console.log(conMayusculas); // a, d, Z

// con objetos

    {edad: 15, nombre: "Felipe"},

usuarios.sort((a, b) => {
    if(a.edad < b.edad) return -1; // a antes de b
    if(a.edad > b.edad) return 1; // b antes de a

});
console.log(usuarios);

// Crear nuevos arrays a partir de un array ya existente
const usuarios = [
    {edad:17, nombre: "Nico"},
    {edad:13, nombre: "Chanchito"},
    {edad:25, nombre: "Felipe"},
    {edad:32, nombre: "Fernanda"},  
];
// Ej: crear un array con los nombres envueltos en la etiquesta HTML de <li> (lista no ordenada) 
const lista = usuarios
.map(usuario => `<li>${usuario.nombre}</li>`) // Recibe una función la cual recibe el elemento que estamos iterando, el indice y el array mismo que esta iterando (solo usaremos el usuario en este)
// El valor que retorne esta función, sera asignado al nuevo array
console.log(lista);
// Un ejemplo del uso de join
const html = `<ol>${lista.join("")}</ol>`;
console.log(html);

// Ej: agregar un elemento de mayor si un usuario tiene mas de 18 años
const mapped = usuarios.map(usuario => ({ // Técnicamente es una línea, asi que sobra los parentesis de llave después de la declaración de los parámetros
    ...usuario, // Una copia de usuario
    mayor: usuario.edad > 17, // un nuevo elemento de mayor
}));
console.log(mapped);

/** Map devuelve un array nuevo, lo que signifca que podemos encadenar métodos como filter y map
 *  El otro método que usaremos se pone después de el elemento que vamos a iterar (en este caso usuarios)
 *  Ej: crear un nuevo array el cual devolvera los nombres de los mayores de edad envueltos en la etiqueta HTMl de <li>
 */

const listaMayores = usuarios
    .filter(u => u.edad > 17)
    .map(u => `<li>${u.nombre}</li>`);

const htmlMayores = `<ol>${listaMayores.join("")}</ol>`;

console.log(htmlMayores);

const numeros = [1, 2, 3, 4];
// Reducir el contenido de un array
// Ej: tomar varios numeros y sumarlos
const suma = numeros.reduce((acc, el) => {
    return acc + el;
    /**
     * acc = 0, el = 1 (return 1)
     * acc = 1, el = 2 (return 3 [futuro acc])
     * acc = 3, el = 3 (return 6)
     * acc = 6, el = 4 (return 10)
     */
}, 0); // recibe una función con dos parametors: un acumulador y el elemento que estamos iterando y también recibe el valor inicial del acumulador
console.log(suma); // 10
// Ej: aplanar un array que contiene mas arrays 
const anidado = [[1, 2], 3, [4, 5]];
// return esperado: [1, 2, 3, 4, 5]
const plano = anidado
    .reduce((acc, el) => 
        acc.concat(el)
    , []);
/**
 * acc = [], el = [1, 2] =concat=> [1, 2]
 * acc = [1, 2], el = 3  =concat=> [1, 2, 3]
 * acc = [1, 2, 3], el = [4, 5] =concat=> [1, 2, 3, 4, 5]
 */
