let i = 0;
// si condicion se cumple
while (i < 10) {
    if (i % 2 == 0) {
        console.log("Número par", i);
    }
    console.log(i); // ejecutar hasta que no se cumpla condición
    i++;
}

>> 07-for-of.js
-------------------------------------------
/* while (x)
 do (x) while (y)
 for (x,x+y,x++) */

let animales = ['Chanchito feliz', 'Dragón', 'Perrito'];

// Como se llamaría cada objeto individual del array y luego el array
for (let animal of animales) {
    console.log(animal);
}

// otra forma de hacerlo en un while
let i = 0;
while (i < animales.length) {
    console.log(animales[i])
    i++;
}

>> 09-break-continue.js
-------------------------------------------
// break para el codigo
// continue se salta partes del codigo
// Aqui se muestra con while, se puede con for, for in, for of, do while

let i = 0;
while (i < 6) {
    i++;
    if (i === 2) {
        continue; // cuando lleguemos aqui, todo lo que este después de este se ignorara y luego empezara otra vez el loop
    }
    if(i === 4) {
        break; // para.
    }
    console.log(i);
}

>> 10-switch.js
-------------------------------------------
// se usa para decirle al codigo, decir donde se vaya dependiendo del valor de una variable

// usa if-else o switch como quieras. Aunque existen librerias que le gustan usar swich como Redux



let accion = 'listar'
// la variable que vamos a evaluar
switch (accion) {
    // donde caiga en el caso de que ...
    case 'listar':
        console.log('Acción de listar');
        break; // poner break al final de todo o si no se ejecutara el resto
    case 'guardar':
        console.log('Acción de guardar');
        break;
    default: // opcional, se ejecuta cuando ningun case a la que la variable pueda responder
        console.log('Acción no reconocida');
}
