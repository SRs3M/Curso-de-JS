/**

 * 8K   7680 x 4320
 * 4K   3040 x 2160
 * WQHD 2560 x 1440
 * FHD  1920 x 1080
 * HD   1280 x 720
 */

function nombreResolucion(ancho, alto) {
        if(ancho >= 1280 || alto >= 720) {
            return "HD";
        } else if(ancho >= 1920 && alto >= 1080) {
            return "FHD";
        } else if(ancho >= 2560 && alto >= 1440) {
            return "WQHD";
        } else if(ancho >= 3040 && alto >= 2160) {
            return "4K";
        } else if(ancho >= 7680 && alto >= 4320) {
            return "8K";

        } else {
            return false;
        }
}

// Crear un algoritmo que devuelva el numero menor y mayor de un array

let array = [2, 5, 7, 15, -5, -100, 55];

function getMenorMayor(arr) {
    let menor = arr[0];
    let mayor = arr[0];
    for (numero of arr) {
        menor = menor < numero ? menor : numero;
        mayor = mayor > numero ? mayor : numero;
    }
    return [menor, mayor];
}

let numeros = getMenorMayor(array);
console.log(numeros);

// Crear un algoritmo que devuelva la cantidad de numeros positivos en un array

let array = [2, 5, 7, 15, -5, -100, 55];

function cuantosPositivos(arr){
    let positivos = [];
    i = 0;
    for(item of arr){
        if(item > 0){
           positivos[i] = item;
           i++
          }
    }
    return positivos;
}

/**
 * crear algoritmo que tome un array de objeto 
 * y devuelva un array de pares
 */
let array = [{
    id: 1,
    name: "Nicolas",
}, {
    id: 2,
    name: "Felipe",
}, {
    id: 3,
    name: "Chanchito",
}];

let pares = [
    [1, {id: 1, name: "Nicolas"}],
    [2, {id: 2, name: "Felipe"}],
    [3, {id: 3, name: "Chanchito"}],
];

function toPairs(arr) {
    let pairs = [];
    for(let idx in arr) {
        let elemento = arr[idx];
        pairs[idx] = [elemento.id, elemento]
    }
    return pairs;
}

let resultado = toPairs(array);
console.log(resultado);

/**
 * crear un array de longitud n, y que sus elementos sean 
 * numeros desde el 1 hasta n
 */

let longitud = 7;

function crearArray(n) {
    let arr = [];
    if(n <= 0){
        return "oe no se vale :(";
    } else {
        for(i = 0; i < n; i++){
            arr[i] = 1 + i;
        }
    }
    return arr;
}

let resultado = crearArray(longitud);
console.log(resultado);
