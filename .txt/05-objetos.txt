// En JS los objetos son dinámicos, es decir, que les podemos agregar o quitar propiedades a nuestro antojo y cambiar su valor en cualquier momento

// Como esto es una constante, no podremos cambiar su valor, ¿O no?
const user = { id:1 }; // Si un objeto tiene pocas propiedades, esta bien colocarlo en una sola línea
// const user = 1; // Esto no se puede hacer ya que estamos cambiandole el valor.. Pero si podemos agregarle propiedades
user.name = "nicolas"; 
user.guardar = function() {
    console.log("guardando", user.name);
}

user.guardar();

delete user.name; //Eliminar propiedades
delete user.guardar;
console.log(user)

// SI queremos que no se pueda modificar
const user1 = Object.freeze({ id:1 }); // No le podremos cambiar ni sus propiedade ni sus valores
user1.name = "nico";
user1.id = 2;
console.log(user1);
const user2 = Object.seal({ id:1 }); // No le podremos cambiar sus valores pero si podemos agregar propiedades
user2.name = "nico";
user2.id = 2;
console.log(user2);

function Usuario() { // El nombre que le daremos es del recurso que queremos crear en UpperCamelCase
    this.id = 1;
    this.recuperarClave = function() { // Método (Una función asignada a una propiedad de un objeto)
        console.log("Recuperando clave...");
    }
}

let usuario = new Usuario;
/** Para crear una variable utilizando el método (Usuario) debemos poner "new", la cual:
 * 1. Creara un objeto literal del aire {}
 * 2. Se vincula el prototipo de la función (En este caso usuario) con la función que se acaba de vincular
 * 3. Este objeto vacio se le asigna a "this." ( Por ende, this = {} )
 * 4. Si esta función de Usuario no retorna nada, retornará de manera automatica "this."

// cuando se define una variable, JS cogera un bloque de memoria y le asignara ese valor de la variable. Pero si creamos un objeto, No se guarda el objeto sino que se guarda la dirección de donde se encuentra el objeto. Asi que aunque tengamos dos objetos totalmente iguales, si hacemos la triple igualdad (a = {}, b = {}, a === b) nos retornara falso, ya que tienen diferentes direcciónes. Pero si queremos que asigne true, deberemos asignar (a = {}, b = a) lo cual definira la misma dirección

let a = 1;
let b = a;
b++;
console.log(a, b); // 1, 2

let x = {};
let y = x;
b.prop = 1;
console.log(x, y); // {}, {}

let n = 1;
function suma(n) { // Cuando creamos un parametro, JS crea una variable especial para ese parametro. Por eso este n es diferente a la variable de n
    n++;
}

suma(n);
console.log(n); // 1

let j = { prop: 1 }
function addition(i) {
    i.prop++;
}

addition(j);
console.log(j); // {prop:2}
